cmake_minimum_required(VERSION 3.20)
project(LyriaScribIA VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LYRIA_BUILD_TESTS "Build tests" ON)
function(lyria_library name)
 add_library(${name} ${ARGN})
 target_include_directories(${name} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/${name}/include)
 target_compile_features(${name} PUBLIC cxx_std_20)
 if(MSVC)
  target_compile_options(${name} PRIVATE /W4)
 else()
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
 endif()
endfunction()
lyria_library(core core/src/core.cpp)
lyria_library(music music/src/music.cpp)
target_link_libraries(music PUBLIC core)
lyria_library(notation notation/src/notation.cpp)
target_link_libraries(notation PUBLIC music)
lyria_library(editor editor/src/editor.cpp)
target_link_libraries(editor PUBLIC music)
lyria_library(midi midi/src/midi.cpp)
target_link_libraries(midi PUBLIC music)
lyria_library(audio audio/src/audio.cpp)
target_link_libraries(audio PUBLIC core)
lyria_library(interpretation interpretation/src/interpretation.cpp)
target_link_libraries(interpretation PUBLIC music)
lyria_library(vocal vocal/src/vocal.cpp)
target_link_libraries(vocal PUBLIC music audio)
lyria_library(choir choir/src/choir.cpp)
target_link_libraries(choir PUBLIC vocal audio)
lyria_library(ai ai/src/ai.cpp)
target_link_libraries(ai PUBLIC music interpretation)
lyria_library(project project/src/project.cpp)
target_link_libraries(project PUBLIC music)
lyria_library(render render/src/render.cpp)
target_link_libraries(render PUBLIC music audio vocal choir interpretation)
lyria_library(plugins plugins/src/plugins.cpp)
target_link_libraries(plugins PUBLIC core)
lyria_library(export export/src/export.cpp)
target_link_libraries(export PUBLIC music)
lyria_library(ui ui/src/ui.cpp)
target_link_libraries(ui PUBLIC editor notation)
add_executable(lyria_scrib app/main.cpp)
target_link_libraries(lyria_scrib PRIVATE core music notation editor midi audio interpretation vocal choir ai project render plugins export ui)
if(LYRIA_BUILD_TESTS)
 enable_testing()
 add_executable(lyria_tests tests/test_main.cpp)
 target_link_libraries(lyria_tests PRIVATE core music notation editor midi audio interpretation vocal choir ai project render plugins export ui)
 add_test(NAME lyria_tests COMMAND lyria_tests)
endif()
